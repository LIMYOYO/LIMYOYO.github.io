\section*{Research Experience}

\entry{Volunteer-Enabled Emergency Medical Device Delivery}{2026--Present}
  {With W. Liu (CUHK), J. Zhang (Cornell) and M. E. H. Ong (NUS and Singapore General Hospital)}
\begin{itemize}
  \item Built a stochastic spatial model comparing fixed-site AED deployment, which needs a two-leg responder trip, with AEDs carried on taxi and ride-hailing fleets, which are limited by fleet idleness.
  \item Derived computable thresholds for when static, mobile or hybrid deployment is optimal, and calibrated the model with more than ten datasets for New York City and Singapore.
\end{itemize}
\vspace{2pt}

\entry{Proactive Dispatching for Semiconductor OHT Systems}{Dec 2025--Present}
  {Supervisor: Prof. Zuo-Jun (Max) Shen \enspace|\enspace The University of Hong Kong \enspace|\enspace Industry partner: TCL}
\begin{itemize}
  \item Developed a proactive empty-vehicle repositioning strategy for overhead hoist transport (OHT) scheduling in semiconductor fabrication, pre-positioning idle vehicles at future demand nodes to reduce system latency.
\end{itemize}
\vspace{2pt}

\entry{Crowd-Sourcing in Rental Network Competition}{Jun 2023--Present}
  {Supervisors: Prof. Ying-Ju Chen and Prof. Yaoming Zhou \enspace|\enspace Shanghai Jiao Tong University}
\begin{itemize}
  \item Built a networked Cournot model of competition under crowd-sourced labor in rental networks, comparing firm, crowd and dual sourcing for firm profits, labor surplus and consumer surplus.
  \item Independently completed model formulation, derivations, coding and manuscript writing.
\end{itemize}
\vspace{2pt}

\entry{E-Bike Allocation Optimization in Competitive Markets}{Sep 2023--Present}
  {Supervisor: Prof. Yaoming Zhou \enspace|\enspace Industry partner: HelloBike}
\begin{itemize}
  \item Developed a reinforcement-learning allocation strategy for a competitive e-bike sharing market, using real-time competitor fleet positions collected by Bluetooth sniffing, with HelloBike (Shanghai) providing data access and validation.
\end{itemize}
\vspace{2pt}

\entry{User-Based Relocation for E-Bike Sharing Systems}{Sep 2023--May 2025}
  {Supervisor: Prof. Yaoming Zhou \enspace|\enspace Shanghai Jiao Tong University}
\begin{itemize}
  \item Designed a user-incentive mechanism with power-level recommendations to complement staff-based relocation; formulated a multi-battery-state Markov model and a variable neighborhood search algorithm, with profit gains of up to 12.6\% on real-world data.
\end{itemize}
\vspace{2pt}

\entry{Staff-Based Relocation and Swapping for E-Bike Sharing Systems}{Sep 2021--Nov 2023}
  {Supervisors: Prof. Jiuh-Biing Sheu and Prof. Yaoming Zhou \enspace|\enspace Shanghai Jiao Tong University}
\begin{itemize}
  \item Integrated e-bike relocation and battery swapping in one Markov-chain model of fleet size and battery levels; proposed an adaptive one-step strategy and a rolling-horizon strategy with over 20\% profit improvement against industrial baselines in simulation on real-world data.
\end{itemize}
